Câu SELECT sinh ra (chỉ đọc)

SELECT "INVOICE_NO", "INVOICE_DATE", "COLLECT_POINT", "TXN_ID", "PIN_NO",
       "SERVICE_CODE", "QUANTITY", "AMOUNT", "INVOICE_STATUS"
FROM   "RECON_SRC"."SRC_FACT_INVOICE_DTP"
WHERE  "INVOICE_DATE" >= :p_from
AND    "INVOICE_DATE" <= :p_to
AND    "ORG_CODE" = :p_org

SQL gộp sinh tự động (chỉ đọc)

SELECT UPPER(TRIM(src."INVOICE_NO"))            AS "K__INVOICE_NO",
       SUM(src."AMOUNT")                        AS "M__TOTAL_AMOUNT",
       SUM(src."QUANTITY")                      AS "M__TOTAL_QTY",
       MIN(TRUNC(src."INVOICE_DATE"))           AS "M__INV_DATE",
       COUNT(DISTINCT TRUNC(src."INVOICE_DATE")) AS "N__INV_DATE",
       COUNT(*)                                 AS "M__LINE_CNT"
FROM  ( <Câu SELECT sinh ra ở bước 3> ) src
GROUP BY UPPER(TRIM(src."INVOICE_NO"))
